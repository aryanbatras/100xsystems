# Docker Code Executor

CloudCode needs to run user code safely. We'll build a Docker-based execution service that isolates each execution in a fresh container, preventing one user's code from affecting others.

## Docker Service

```java
// src/main/java/com/cloudcode/execution/DockerService.java
package com.cloudcode.execution;

import com.github.dockerjava.api.DockerClient;
import com.github.dockerjava.api.command.CreateContainerResponse;
import com.github.dockerjava.api.model.*;
import com.github.dockerjava.core.DockerClientBuilder;
import org.springframework.stereotype.Service;

import java.io.ByteArrayOutputStream;
import java.util.List;

@Service
public class DockerService {

    private final DockerClient dockerClient;

    public DockerService() {
        this.dockerClient = DockerClientBuilder.getInstance().build();
    }

    public ExecutionResult executeCode(
            String code,
            String language,
            int timeoutSeconds) {

        String image = getImageForLanguage(language);
        String containerId = null;

        try {
            // Pull image if not present
            dockerClient.pullImageCmd(image).start().awaitCompletion();

            // Create container
            CreateContainerResponse container = dockerClient
                .createContainerCmd(image)
                .withCmd("sh", "-c", code)
                .withHostConfig(HostConfig.newHostConfig()
                    .withMemory(256 * 1024 * 1024L)      // 256MB limit
                    .withMemorySwap(0L)                    // No swap
                    .withCpuCount(1L)                      // 1 CPU core
                    .withReadonlyRootfs(true)              // Read-only filesystem
                    .withAutoRemove(true))                 // Auto cleanup
                .withNetworkDisabled(true)                 // No network
                .exec();

            containerId = container.getId();

            // Start container
            dockerClient.startContainerCmd(containerId).exec();

            // Wait for completion with timeout
            dockerClient.waitContainerCmd(containerId)
                .exec(new WaitContainerResultCallback())
                .awaitStatusCode(timeoutSeconds, java.util.concurrent.TimeUnit.SECONDS);

            // Get logs
            ByteArrayOutputStream stdout = new ByteArrayOutputStream();
            ByteArrayOutputStream stderr = new ByteArrayOutputStream();

            dockerClient.logContainerCmd(containerId)
                .withStdOut(true)
                .withStdErr(true)
                .exec(new LogContainerResultCallback())
                .awaitCompletion();

            return new ExecutionResult(
                stdout.toString(),
                stderr.toString(),
                true,
                null
            );

        } catch (Exception e) {
            return new ExecutionResult(
                null,
                null,
                false,
                "Execution failed: " + e.getMessage()
            );
        } finally {
            // Cleanup
            if (containerId != null) {
                try {
                    dockerClient.removeContainerCmd(containerId)
                        .withForce(true)
                        .exec();
                } catch (Exception ignored) {}
            }
        }
    }

    private String getImageForLanguage(String language) {
        return switch (language.toLowerCase()) {
            case "java" -> "openjdk:21-slim";
            case "python" -> "python:3.12-slim";
            case "typescript", "javascript" -> "node:20-slim";
            default -> "alpine:latest";
        };
    }
}
```
